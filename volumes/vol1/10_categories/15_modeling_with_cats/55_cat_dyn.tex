\begin{ctdefinition}[Endofunctions]
    \label{def:cat-endofunctions}
    The category $\EndSet$ of endofunctions is:
    \begin{enumerate}
        \item \emph{Objects}: pairs $\tup{\setA, \stylemorph{\varphi}}$ where $\setA$ is a set and $\stylemorph{\varphi} \colon \setA \to \setA$ is a function.
        \item \emph{Morphisms}: a morphism $\mora \colon \tup{\setA, \stylemorph{\varphi}} \mto_\EndSet \tup{\setB, \stylemorph{\psi}}$ is a function $\mora \colon \setA \to_\Set \setB$ with the property that $\mora \mthen \stylemorph{\psi}= \stylemorph{\varphi} \mthen \mora$.
        \item \emph{Composition}: the usual composition of functions.
        \item \emph{Identity morphisms}: identity functions.
    \end{enumerate}
\end{ctdefinition}

\subsection{Linear time-invariant systems}
\label{sec:cat-dyn-lti}

In \cref{sec:dynamical-systems} we met \emph{linear time-invariant (LTI) systems}: dynamical systems which, in addition to a state, have inputs and outputs (\cref{def:LTI_syst_first_time}).
We now show that continuous-time LTI systems are the morphisms of a category, in which composition models connecting two systems in series.

Recall that such a system receives an input signal $\mat{u}(t) \setin \reals^n$, maintains an internal state $\mat{x}(t) \setin \reals^k$, and produces an output signal $\mat{y}(t) \setin \reals^m$.
The state starts at a given start state (or initial state) $\prstart \setin \reals^k$, and the state and the output evolve according to the equations
\begin{equation}
    \label{eq:cat-dyn-lti-equations}
    \begin{aligned}
        \dot{\mat{x}}(t) & = \mat{A}\mat{x}(t) + \mat{B}\mat{u}(t), \\
        \mat{y}(t)       & = \mat{C}\mat{x}(t) + \mat{D}\mat{u}(t),
    \end{aligned}
\end{equation}
where $\mat{A}$, $\mat{B}$, $\mat{C}$, and $\mat{D}$ are real matrices.
The key idea is to view an LTI system as a process that turns signals in its input space $\reals^n$ into signals in its output space $\reals^m$, that is, as a morphism from $\reals^n$ to $\reals^m$.
Two such systems can be composed in series, by feeding the output of the first into the second as its input, whenever the output space of the first is the input space of the second.

Throughout, we think of the elements of $\reals^k$ as column vectors, that is, as matrices with $k$ rows and one column.

\begin{ctdefinition}[Category \LTI]
    \label{def:LTICat}
    \SYNDEF{category of finite-dimensional linear time-invariant systems}
    The category \LTI of LTI systems is the category given by:

    \constit

    \begin{enumerate}
        \item Objects: the real coordinate spaces $\reals^n$, for $n \setin \natnumbers$.
        \item Morphisms: a morphism from $\reals^n$ to $\reals^m$ is an LTI system with input space $\reals^n$ and output space $\reals^m$, by which we mean a tuple
              \begin{equation}
                  \tupp{\reals^k, \prstart, \mat{A}, \mat{B}, \mat{C}, \mat{D}},
              \end{equation}
              where $\reals^k$, for some $k \setin \natnumbers$, is the \emph{state space}, $\prstart \setin \reals^k$ is the \emph{start state}, and
              \begin{equation}
                  \mat{A} \setin \reals^{\matdim{k}{k}}, \quad
                  \mat{B} \setin \reals^{\matdim{k}{n}}, \quad
                  \mat{C} \setin \reals^{\matdim{m}{k}}, \quad
                  \mat{D} \setin \reals^{\matdim{m}{n}}
              \end{equation}
              are real matrices.
        \item Composition: given morphisms $\mora \colon \reals^n \mto \reals^m$ and $\morb \colon \reals^m \mto \reals^p$, described by the LTI systems
              \begin{equation}
                  \begin{aligned}
                      \mora & = \tupp{\reals^k, \prstart_\mora, \mat{A}_\mora, \mat{B}_\mora, \mat{C}_\mora, \mat{D}_\mora}, \\
                      \morb & = \tupp{\reals^l, \prstart_\morb, \mat{A}_\morb, \mat{B}_\morb, \mat{C}_\morb, \mat{D}_\morb},
                  \end{aligned}
              \end{equation}
              their composition $(\morab) \colon \reals^n \mto \reals^p$ is the LTI system $\tupp{\reals^{k+l}, \prstart, \mat{A}, \mat{B}, \mat{C}, \mat{D}}$, where
              \begin{equation}
                  \label{eq:LTI-composition}
                  \begin{aligned}
                      \prstart & =\begin{bmatrix}
                                      \prstart_\mora \\
                                      \prstart_\morb \\
                                  \end{bmatrix},\quad
                      \mat{A}=\begin{bmatrix}
                                  \mat{A}_\mora              & \mat{0}       \\
                                  \mat{B}_\morb\mat{C}_\mora & \mat{A}_\morb
                              \end{bmatrix},\quad
                      \mat{B}=\begin{bmatrix}
                                  \mat{B}_\mora \\
                                  \mat{B}_\morb\mat{D}_\mora
                              \end{bmatrix}, \\
                      \mat{C}  & =\begin{bmatrix}
                                      \mat{D}_\morb\mat{C}_\mora & \mat{C}_\morb
                                  \end{bmatrix}, \quad
                      \mat{D}=\mat{D}_\morb\mat{D}_\mora.
                  \end{aligned}
              \end{equation}
        \item Identity morphisms: the identity morphism of $\reals^n$ is the LTI system
              \begin{equation}
                  \tupp{\reals^0, \mat{0}^{\matdim{0}{1}},\mat{0}^{\matdim{0}{0}},\mat{0}^{\matdim{0}{n}},\mat{0}^{\matdim{n}{0}},\mat{1}^{\matdim{n}{n}}},
              \end{equation}
              where $\mat{1}^{\matdim{n}{n}}$ denotes the $n \times n$ identity matrix, and the remaining entries are the (unique) matrices of the indicated dimensions, which have no entries.
    \end{enumerate}
\end{ctdefinition}

\begin{remark}[Where the composition rule comes from]
    The matrices in \cref{eq:LTI-composition} are obtained by writing down the equations \cref{eq:cat-dyn-lti-equations} for $\mora$ and for $\morb$, and then substituting the output of $\mora$ for the input of $\morb$.
    Writing $\mat{x}_\mora$ and $\mat{x}_\morb$ for the states of the two systems, and $\mat{u}$ for the input of $\mora$, this gives
    \begin{equation}
        \begin{aligned}
            \dot{\mat{x}}_\mora(t) & = \mat{A}_\mora\mat{x}_\mora(t) + \mat{B}_\mora\mat{u}(t), \\
            \dot{\mat{x}}_\morb(t) & = \mat{B}_\morb\mat{C}_\mora\mat{x}_\mora(t) + \mat{A}_\morb\mat{x}_\morb(t) + \mat{B}_\morb\mat{D}_\mora\mat{u}(t), \\
            \mat{y}(t)             & = \mat{D}_\morb\mat{C}_\mora\mat{x}_\mora(t) + \mat{C}_\morb\mat{x}_\morb(t) + \mat{D}_\morb\mat{D}_\mora\mat{u}(t),
        \end{aligned}
    \end{equation}
    where $\mat{y}$ is the output of $\morb$.
    These are again equations of the form \cref{eq:cat-dyn-lti-equations}, for the stacked state vector with $\mat{x}_\mora(t)$ on top of $\mat{x}_\morb(t)$.
\end{remark}

\begin{remark}[The state space of a composite]
    \label{rem:LTI-state-space-trick}
    The state of the composite system $\morab$ consists of a state of $\mora$ together with a state of $\morb$.
    It would therefore be natural to take the \SY{cartesian product} $\reals^k \cartprod \reals^l$ as the state space of $\morab$.
    We deliberately do not do this.
    Instead, we take the coordinate space $\reals^{k+l}$, whose dimension is the sum of the two dimensions, and whose elements are obtained by stacking a column vector with $k$ rows on top of a column vector with $l$ rows.

    The reason is that the elements of $\reals^k \cartprod \reals^l$ are pairs of vectors, and so for three composable systems the state spaces $(\reals^k \cartprod \reals^l) \cartprod \reals^p$ and $\reals^k \cartprod (\reals^l \cartprod \reals^p)$ consist of differently nested pairs: they are isomorphic, but not equal.
    In contrast,
    \begin{equation}
        \reals^{(k+l)+p} = \reals^{k+(l+p)}
        \qquad \text{and} \qquad
        \reals^{k+0} = \reals^k = \reals^{0+k}
    \end{equation}
    hold on the nose, simply because addition of natural numbers is \SY{associative} and has $0$ as neutral element.
    This is what makes composition in \LTI \SY{associative} and unital exactly, and not only ``up to an isomorphism''.
    We will meet the same issue again in \cref{sec:cat-machines} for \SY{Moore machines}, whose state spaces are arbitrary sets, where no such simple trick is available.
\end{remark}

In what follows we often leave the state space implicit, and write a morphism of \LTI simply as $\genericlti{}$.

\begin{exercise}
    \label{ex:LTI-is-a-category}
    Prove that \LTI is indeed a category.
\end{exercise}
\begin{solution}
    We start with associativity.
    Consider morphism $\mora \colon \reals^l\mto \reals^m$ given by~$\genericlti{\mora}$, and $\morb\colon \reals^m\mto \reals^n$ given by~$\genericlti{\morb}$, and $\morc\colon \reals^n\mto \reals^o$ given by~$\genericlti{\morc}$.
    The morphism~$\morab$ is described by the LTI
    \begin{equation}
        \genericlti{\mora,\morb},
    \end{equation}
    where
    \begin{widepar}
        \begin{equation}
            \prstart_{\mora,\morb}=\begin{bmatrix}
                \prstart_\mora \\
                \prstart_\morb
            \end{bmatrix},\quad
            \mat{A}_{\mora,\morb}=\begin{bmatrix}
                \mat{A}_\mora              & \mat{0}       \\
                \mat{B}_\morb\mat{C}_\mora & \mat{A}_\morb
            \end{bmatrix},\
            \mat{B}_{\mora,\morb}=\begin{bmatrix}
                \mat{B}_\mora \\
                \mat{B}_\morb\mat{D}_\mora
            \end{bmatrix},\
            \mat{C}_{\mora,\morb}=\begin{bmatrix}
                \mat{D}_\morb\mat{C}_\mora & \mat{C}_\morb
            \end{bmatrix}, \
            \mat{D}_{\mora,\morb}=\mat{D}_\morb\mat{D}_\mora.
        \end{equation}
    \end{widepar}
    The morphism~$(\morab)\mthen \morc$ is described by
    \begin{equation}
        \genericlti{(\mora,\morb),\morc},
    \end{equation}
    where:
    \begin{widepar}
        \begin{equation}
            \label{eq:assoc_lti_1}
            \begin{aligned}
                \prstart_{(\mora,\morb),\morc} & =\begin{bmatrix}
                                                      \prstart_\mora \\
                                                      \prstart_\morb \\
                                                      \prstart_\morc
                                                  \end{bmatrix},\quad
                \mat{A}_{(\mora,\morb),\morc}=\begin{bmatrix}
                                                  \mat{A}_\mora                           & \mat{0}                    & \mat{0}       \\
                                                  \mat{B}_\morb\mat{C}_\mora              & \mat{A}_\morb              & \mat{0}       \\
                                                  \mat{B}_\morc\mat{D}_\morb\mat{C}_\mora & \mat{B}_\morc\mat{C}_\morb & \mat{A}_\morc
                                              \end{bmatrix},\
                \mat{B}_{(\mora,\morb),\morc}=\begin{bmatrix}
                                                  \mat{B}_\mora              \\
                                                  \mat{B}_\morb\mat{D}_\mora \\
                                                  \mat{B}_\morc\mat{D}_\morb\mat{D}_\mora
                                              \end{bmatrix}, \\
                \mat{C}_{(\mora,\morb),\morc}  & =\begin{bmatrix}
                                                      \mat{D}_\morc\mat{D}_\morb\mat{C}_\mora & \mat{D}_\morc\mat{C}_\morb & \mat{C}_\morc
                                                  \end{bmatrix}, \
                \mat{D}_{(\mora,\morb),\morc}=\mat{D}_\morc\mat{D}_\morb\mat{D}_\mora.
            \end{aligned}
        \end{equation}
    \end{widepar}
    On the other hand, the morphism~$\morb\mthen \morc$ is described by
    \begin{equation}
        \genericlti{\morb,\morc},
    \end{equation}
    where:
    \begin{widepar}
        \begin{equation}
            \prstart_{\morb,\morc}=\begin{bmatrix}
                \prstart_\morb \\
                \prstart_\morc
            \end{bmatrix},\
            \mat{A}_{\morb,\morc}=\begin{bmatrix}
                \mat{A}_\morb              & \mat{0}       \\
                \mat{B}_\morc\mat{C}_\morb & \mat{A}_\morc
            \end{bmatrix},\
            \mat{B}_{\morb,\morc}=\begin{bmatrix}
                \mat{B}_\morb \\
                \mat{B}_\morc\mat{D}_\morb
            \end{bmatrix},\
            \mat{C}_{\morb,\morc}=\begin{bmatrix}
                \mat{D}_\morc\mat{C}_\morb & \mat{C}_\morc
            \end{bmatrix}, \
            \mat{D}_{\morb,\morc}=\mat{D}_\morc\mat{D}_\morb.
        \end{equation}
    \end{widepar}
    Furthermore, the morphism~$\mora\mthen (\morb\mthen \morc)$ is described by
    \begin{equation}
        \genericlti{\mora,(\morb,\morc)},
    \end{equation}
    where:
    \begin{widepar}
        \begin{equation}
            \begin{aligned}
                \label{eq:assoc_lti_2}
                \prstart_{\mora,(\morb,\morc)} & =\begin{bmatrix}
                                                      \prstart_\mora \\
                                                      \prstart_\morb \\
                                                      \prstart_\morc
                                                  \end{bmatrix},\quad
                \mat{A}_{\mora,(\morb,\morc)}=\begin{bmatrix}
                                                  \mat{A}_\mora                           & \mat{0}                    & \mat{0}       \\
                                                  \mat{B}_\morb\mat{C}_\mora              & \mat{A}_\morb              & \mat{0}       \\
                                                  \mat{B}_\morc\mat{D}_\morb\mat{C}_\mora & \mat{B}_\morc\mat{C}_\morb & \mat{A}_\morc
                                              \end{bmatrix},\
                \mat{B}_{\mora,(\morb,\morc)}=\begin{bmatrix}
                                                  \mat{B}_\mora              \\
                                                  \mat{B}_\morb\mat{D}_\mora \\
                                                  \mat{B}_\morc\mat{D}_\morb\mat{D}_\mora
                                              \end{bmatrix}, \\
                \mat{C}_{\mora,(\morb,\morc)}  & =\begin{bmatrix}
                                                      \mat{D}_\morc\mat{D}_\morb\mat{C}_\mora & \mat{D}_\morc\mat{C}_\morb & \mat{C}_\morc
                                                  \end{bmatrix}, \
                \mat{D}_{\mora,(\morb,\morc)}=\mat{D}_\morc\mat{D}_\morb\mat{D}_\mora.
            \end{aligned}
        \end{equation}
    \end{widepar}
    Clearly, the matrices in \cref{eq:assoc_lti_1} and \cref{eq:assoc_lti_2} coincide.
    The state spaces coincide as well: if $\reals^{k_\mora}$, $\reals^{k_\morb}$, and $\reals^{k_\morc}$ are the state spaces of $\mora$, $\morb$, and $\morc$, then the state space of $(\morab)\mthen \morc$ is $\reals^{(k_\mora+k_\morb)+k_\morc}$ and that of $\mora\mthen (\morb\mthen \morc)$ is $\reals^{k_\mora+(k_\morb+k_\morc)}$, and these are equal because addition of natural numbers is associative (\cref{rem:LTI-state-space-trick}).
    This shows associativity.

    We now show unitality.
    Consider a morphism~$\mora\colon \reals^l\mto \reals^m$, described by~$\genericlti{}$, with state space $\reals^k$.
    The morphism~$\catidat{\reals^l} \mthen \mora$ is a morphism~$\reals^l\mto \reals^m$ still given by~$\genericlti{}$.
    Similarly, the morphism~$\mora\mthen \catidat{\reals^m}$ is a morphism~$\reals^l\mto \reals^m$, given by~$\genericlti{}$.
    In both cases the state space is again $\reals^k$, since $\reals^{0+k} = \reals^k = \reals^{k+0}$.
    Therefore, \LTI is a category.
\end{solution}

\begin{example}[Composing two LTI systems]
    \label{exa:lti-composition-spring-mass}
    Consider the mass between two springs of \cref{exa:sping-mass-lti}.
    It is a morphism $\mora \colon \reals^1 \mto \reals^1$ of \LTI, whose input is the force $F$, whose output is the position $q$ of the mass, and whose state space is $\reals^2$; its matrices are
    \begin{equation*}
        \mat{A}_\mora=\begin{bmatrix}0                  & 1 \\
               -\frac{k_1+k_2}{m} & 0\end{bmatrix},\quad
        \mat{B}_\mora=\begin{bmatrix}
            0 \\
            \frac{1}{m}
        \end{bmatrix},
        \quad
        \mat{C}_\mora=\begin{bmatrix}
            1 & 0
        \end{bmatrix},
        \quad
        \mat{D}_\mora=\begin{bmatrix}
            0
        \end{bmatrix}.
    \end{equation*}
    Consider also the LTI system $\morb \colon \reals^1 \mto \reals^1$ with a one-dimensional state $p$, given by
    \begin{equation*}
        \begin{aligned}
            \dot{p}(t) & =p(t)+c\,q(t), \\
            w(t)       & =p(t),
        \end{aligned}
    \end{equation*}
    which takes as input the output produced by the spring-mass system (the position of the mass along the horizontal dimension), scaled by a constant factor $c$.
    Its matrices are $\mat{A}_\morb = \begin{bmatrix} 1 \end{bmatrix}$, $\mat{B}_\morb = \begin{bmatrix} c \end{bmatrix}$, $\mat{C}_\morb = \begin{bmatrix} 1 \end{bmatrix}$, and $\mat{D}_\morb = \begin{bmatrix} 0 \end{bmatrix}$.

    We can compose the two systems.
    By \cref{eq:LTI-composition}, the composite $\morab \colon \reals^1 \mto \reals^1$ has the state space $\reals^3$ and the matrices
    \begin{equation*}
        \mat{A}=\begin{bmatrix}0                  & 1 & 0 \\
               -\frac{k_1+k_2}{m} & 0 & 0 \\
               c                  & 0 & 1\end{bmatrix},\quad
        \mat{B}=\begin{bmatrix}
            0 \\
            \frac{1}{m} \\
            0
        \end{bmatrix},
        \quad
        \mat{C}=\begin{bmatrix}
            0 & 0 & 1
        \end{bmatrix},
        \quad
        \mat{D}=\begin{bmatrix}
            0
        \end{bmatrix}.
    \end{equation*}
    This can be visualized more intuitively with the explicit composed dynamics:
    \begin{equation*}
        \begin{aligned}
            \begin{bmatrix}
                \dot{q}(t)  \\
                \ddot{q}(t) \\
                \dot{p}(t)
            \end{bmatrix} & =
            \begin{bmatrix}0                  & 1 & 0 \\
               -\frac{k_1+k_2}{m} & 0 & 0 \\
               c                  & 0 & 1\end{bmatrix} \begin{bmatrix}
                                                        {q}(t)     \\
                                                        \dot{q}(t) \\
                                                        {p}(t)
                                                    \end{bmatrix}+
            \begin{bmatrix}
                0 \\
                \frac{1}{m} \\
                0
            \end{bmatrix} F \\
            w(t)            & =\begin{bmatrix}
                                   0 & 0 & 1
                               \end{bmatrix}\begin{bmatrix}
                                                {q}(t)     \\
                                                \dot{q}(t) \\
                                                {p}(t)
                                            \end{bmatrix}
        \end{aligned}
    \end{equation*}
    Given an input force, we will get a transformed horizontal position as an output.
\end{example}

\paragraph{Equivalent LTI systems}
The same system can be specified by different tuples, which differ only by a change of coordinates of the state space (compare \cref{exa:lti-similarity-transformation}).

\begin{definition}[Equivalence of LTI systems]
    \label{def:equivalence_lti}
    \SYNDEF{equivalence of LTI systems}
    Two LTI systems~$\genericlti{\mora}$ and~$\genericlti{\morb}$ with the same input space, the same output space, and state spaces of the same dimension $k$ are \emph{equivalent} if and only if there exists an invertible matrix $\mat{T} \setin \reals^{\matdim{k}{k}}$ such that
    \begin{equation}
        \prstart_\morb=\mat{T}\prstart_\mora, \ \mat{A}_\morb = \mat{T}\mat{A}_\mora \mat{T}^{-1}, \ \mat{B}_\morb = \mat{T}\mat{B}_\mora, \ \mat{C}_\morb =\mat{C}_\mora\mat{T}^{-1},\ \mat{D}_\morb=\mat{D}_\mora.
    \end{equation}
    $\mat{T}$ is called an \emph{equivalence transformation}; it relates the states of the two systems by $\mat{x}_\morb(t)=\mat{T}\mat{x}_\mora(t)$.
\end{definition}

We think of equivalent LTI-systems as different ways of specifying what is essentially ``the same system''.
What is different in each specification is only different by a change of coordinates.
