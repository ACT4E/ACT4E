\begin{gradedsolution}
\begin{enumerate}
    \item We need to show that for all $\posela, \poselb \setin \posAset$, if $\posela \setin \subA_{\mapa}$ and $\posela \posAleq \poselb$, then $\poselb \setin \subA_{\mapa}$.
    So let $\posela \setin \subA_{\mapa}$, and let $\poselb \setin \posAset$ with $\posela \posAleq \poselb$.
    Since, by assumption, $\mapa$ is monotone, we find that $\mapa(\posela) \leq \mapa(\poselb)$.
    Since $\mapa(\posela) = \true$ and $\true$ is the greatest element of $\Bool$, it follows that $\mapa(\poselb) = \true$.
    This means that $\poselb \setin \subA_{\mapa}$, as was to be shown.

    \item Let $\subA \setsubseteq \posAset$ be an upper set.

    \emph{Uniqueness.}
    Suppose $\mapa \colon \posA \mto \Bool$ satisfies $\subA_{\mapa} = \subA$.
    Then $\mapa(\posAel) = \true$ for all $\posAel \setin \subA$, and, since $\Bool$ has only the two elements $\true$ and $\false$, $\mapa(\posAel) = \false$ for all $\posAel \notin \subA$.
    So $\mapa$ is completely determined by $\subA$: it must be the indicator function of $\subA$.

    \emph{Existence.}
    Define $\mapa \colon \posAset \sto \Bool$ to be the indicator function of $\subA$:
    \begin{equation}
        \mapa(\posAel) =
        \begin{cases}
            \true  & \text{ if } \posAel \setin \subA, \\
            \false & \text{ if } \posAel \notin \subA.
        \end{cases}
    \end{equation}
    By construction, $\subA_{\mapa} = \subA$, so it remains to check that $\mapa$ is monotone.
    Let $\posela, \poselb \setin \posAset$ with $\posela \posAleq \poselb$.
    If $\mapa(\posela) = \false$, then $\mapa(\posela) \leq \mapa(\poselb)$ holds because $\false$ is the least element of $\Bool$.
    If $\mapa(\posela) = \true$, then $\posela \setin \subA$; since $\subA$ is an upper set, also $\poselb \setin \subA$, hence $\mapa(\poselb) = \true$, and again $\mapa(\posela) \leq \mapa(\poselb)$.
\end{enumerate}
Taken together, the two parts show that $\mapa \mapsto \subA_{\mapa}$ is a 1-to-1 correspondence between monotone maps $\posA \mto \Bool$ and upper sets of $\posA$.
\end{gradedsolution}
