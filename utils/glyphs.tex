\newcommand{\uparrowcircle}{%
  \tikz[baseline=-0.60ex]{
    \node[inner sep=0pt, outer sep=0pt] at (0,0) {$\uparrow$};
    \draw[fill=white,line width=0.6pt] (0,-0.75ex) circle (0.25ex);
  }%
}

  }%
}
